function renderGoods(){

  let data = [];

  if(curSort==='newest') data.sort((a,b)=>b.id-a.id);

  else if(curSort==='cheap'){

    data.sort((a,b)=>{

      const pa = a.type==='demand' ? (a.budgetMin||0) : (a.price||0);

      const pb = b.type==='demand' ? (b.budgetMin||0) : (b.price||0);

      return pa-pb;

    });

  } else if(curSort==='expensive'){

    data.sort((a,b)=>{

      const pa = a.type==='demand' ? (a.budgetMax||a.budgetMin||0) : (a.price||0);

      return pb-pa;

    });

  } else if(curSort==='hot') data.sort((a,b)=>(b.views||0)-(a.views||0));

  else data.sort((a,b)=>(b.tag?1:0)-(a.tag?1:0)||b.id-a.id);

  const wf = document.getElementById('waterfall');

}

function applyFilter(){

  const cat = document.querySelector('#fCat .fopt.on')?.textContent||'全部';

  const cond = document.querySelector('#fCond .fopt.on')?.textContent||'不限';

  const trade = document.querySelector('#fTrade .fopt.on')?.textContent||'不限';

  const pmin = document.getElementById('fPriceMin').value.trim();

  const pmax = document.getElementById('fPriceMax').value.trim();

 

  let data = [...DATA];

  if(cat!=='全部') data = data.filter(g=>g.cat===cat);

  if(cond!=='不限'){

    if(cond==='七成新及以下') data = data.filter(g=>['七成新','六成新及以下'].includes(g.cond));

    else data = data.filter(g=>g.cond===cond);

  }

  if(trade!=='不限') data = data.filter(g=>g.trade===trade);

  if(pmin) data = data.filter(g=>g.price>=+pmin);

  if(pmax) data = data.filter(g=>g.price<=+pmax);

 

  document.getElementById('glist').innerHTML = data.length===0 ? '<div class="empty"><div class="ei">��</div><div class="et">没有找到符合条件的商品</div></div>' : data.slice(0,6).map(g=>card(g)).join('');

  document.getElementById('waterfall').innerHTML = data.length<=6 ? '' : data.slice(6).map(g=>card(g)).join('');

  closeFilter();

  showPage('home');

  showToast(`✅ 筛选完成，共 ${data.length} 件`);

}
